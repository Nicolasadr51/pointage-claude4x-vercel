<script>
// Configuration globale
const API_URL = '/api';
let dataTable;

// Fonction de sécurité
function sanitize(str) {
    return DOMPurify.sanitize(str);
}

// Gestion du loading
function showLoading() {
    document.getElementById('loadingSpinner').style.display = 'flex';
}

function hideLoading() {
    document.getElementById('loadingSpinner').style.display = 'none';
}

// Initialisation des graphiques
function initCharts() {
    // Graphique des pointages quotidiens
    const ctxDaily = document.getElementById('dailyCheckinsChart').getContext('2d');
    new Chart(ctxDaily, {
        type: 'line',
        data: {
            labels: ['Lun', 'Mar', 'Mer', 'Jeu', 'Ven', 'Sam', 'Dim'],
            datasets: [{
                label: 'Pointages',
                data: [12, 19, 15, 17, 14, 8, 5],
                borderColor: 'rgb(75, 192, 192)',
                tension: 0.1
            }]
        }
    });

    // Graphique des heures travaillées
    const ctxHours = document.getElementById('workedHoursChart').getContext('2d');
    new Chart(ctxHours, {
        type: 'bar',
        data: {
            labels: ['Lun', 'Mar', 'Mer', 'Jeu', 'Ven', 'Sam', 'Dim'],
            datasets: [{
                label: 'Heures',
                data: [8, 7.5, 8, 8.5, 7, 4, 0],
                backgroundColor: 'rgba(54, 162, 235, 0.2)',
                borderColor: 'rgb(54, 162, 235)',
                borderWidth: 1
            }]
        }
    });
}

// Initialisation de la DataTable
function initDataTable() {
    dataTable = $('#pointagesTable').DataTable({
        ajax: {
            url: `${API_URL}/pointages`,
            dataSrc: ''
        },
        columns: [
            { 
                data: null,
                render: function() {
                    return '<input type="checkbox" class="checkin-select">';
                }
            },
            { data: 'id' },
            { data: 'employee_name' },
            { data: 'date' },
            { data: 'time_in' },
            { data: 'time_out' },
            {
                data: null,
                render: function(data) {
                    return `
                        <button class="btn btn-sm btn-warning edit-btn" data-id="${data.id}">
                            <i class="fas fa-edit"></i>
                        </button>
                        <button class="btn btn-sm btn-danger delete-btn" data-id="${data.id}">
                            <i class="fas fa-trash"></i>
                        </button>
                    `;
                }
            }
        ],
        order: [[1, 'desc']],
        language: {
            url: '//cdn.datatables.net/plug-ins/1.13.7/i18n/fr-FR.json'
        }
    });
}

// Gestionnaires d'événements
function initEventListeners() {
    // Sauvegarde du pointage
    $('#btnSave').click(async function() {
        if (!$('#pointageForm')[0].checkValidity()) {
            return;
        }

        const pointageData = {
            employee_id: $('#employee').val(),
            date: $('#date').val(),
            time_in: $('#timeIn').val(),
            time_out: $('#timeOut').val()
        };

        try {
            showLoading();
            const id = $('#pointageId').val();
            const method = id ? 'PUT' : 'POST';
            const url = id ? `${API_URL}/pointages/${id}` : `${API_URL}/pointages`;

            const response = await fetch(url, {
                method,
                headers: {
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify(pointageData)
            });

            if (!response.ok) throw new Error('Erreur serveur');

            $('#pointageModal').modal('hide');
            dataTable.ajax.reload();
            showAlert('Pointage enregistré avec succès', 'success');
        } catch (error) {
            showAlert('Erreur lors de l\'enregistrement', 'danger');
        } finally {
            hideLoading();
        }
    });

    // Export des données
    $('#btnExport').click(async function() {
        try {
            showLoading();
            const response = await fetch(`${API_URL}/export`);
            const blob = await response.blob();
            const url = window.URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = 'pointages.xlsx';
            document.body.appendChild(a);
            a.click();
            window.URL.revokeObjectURL(url);
        } catch (error) {
            showAlert('Erreur lors de l\'export', 'danger');
        } finally {
            hideLoading();
        }
    });

    // Filtres
    $('#filterEmployee, #filterStartDate, #filterEndDate').change(function() {
        dataTable.ajax.reload();
    });
}

// Fonction d'affichage des alertes
function showAlert(message, type) {
    const alert = `
        <div class="alert alert-${type} alert-dismissible fade show" role="alert">
            ${sanitize(message)}
            <button type="button" class="btn-close" data-bs-dismiss="alert"></button>
        </div>
    `;
    
    const alertContainer = document.createElement('div');
    alertContainer.style.position = 'fixed';
    alertContainer.style.top = '20px';
    alertContainer.style.right = '20px';
    alertContainer.style.zIndex = '9999';
    alertContainer.innerHTML = alert;
    
    document.body.appendChild(alertContainer);
    
    setTimeout(() => {
        alertContainer.remove();
    }, 3000);
}

// Initialisation
document.addEventListener('DOMContentLoaded', function() {
    initCharts();
    initDataTable();
    initEventListeners();
});
</script>
